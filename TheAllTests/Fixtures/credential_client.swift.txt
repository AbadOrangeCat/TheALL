import Foundation
import Security

enum L10n { static func string(_ key: String, _ arguments: CVarArg...) -> String { key } }
struct EmptyLegacy: AICredentialStore {
    func load(for provider: AIProvider) throws -> AIConnection? { nil }
    func save(_ connection: AIConnection, for provider: AIProvider) throws { fatalError("No legacy writes") }
    func delete(for provider: AIProvider) throws { fatalError("No legacy deletes") }
}

let arguments = CommandLine.arguments
let defaults = UserDefaults(suiteName: arguments[3])!
defer { defaults.removePersistentDomain(forName: arguments[3]) }
let store = PersistentAICredentialStore(legacy: EmptyLegacy(), defaults: defaults,
    installationRoot: URL(fileURLWithPath: arguments[2]), keychainPath: arguments[1])
let fake = AIConnection(apiKey: "fake-cross-build-key", baseURL: "https://fixture.invalid", model: "fixture")
do {
    switch arguments[4] {
    case "save": try store.save(fake, for: .claude)
    case "load": guard try store.load(for: .claude) == fake else { exit(2) }
    case "delete": try store.delete(for: .claude)
    case "absent": guard try store.load(for: .claude) == nil else { exit(3) }
    case "raw":
        // Bypass the app-side check to test the helper's parent authentication.
        let process = Process(); process.executableURL = URL(fileURLWithPath: arguments[5]); process.environment = [:]
        let input = Pipe(), output = Pipe()
        process.standardInput = input; process.standardOutput = output; process.standardError = FileHandle.nullDevice
        try process.run()
        let request = AICredentialHelperRequest(operation: .load, provider: "claude", keychainPath: arguments[1])
        try? input.fileHandleForWriting.write(contentsOf: JSONEncoder().encode(request))
        try? input.fileHandleForWriting.close()
        let response = try JSONDecoder().decode(AICredentialHelperResponse.self, from: output.fileHandleForReading.readDataToEndOfFile())
        process.waitUntilExit()
        guard response.status != errSecSuccess, response.data == nil else { exit(4) }
    default: exit(5)
    }
    #if CLIENT_SECOND
    print("client-two success")
    #else
    print("client-one success")
    #endif
} catch { print("fixture failed: \(error)"); exit(1) }
